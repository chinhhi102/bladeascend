# Nguồn SỰ THẬT DUY NHẤT cho mọi message đường nóng — NETCODE_SPEC §4.9.
#
# Vì sao bắt buộc có file này TRƯỚC khi bật Protocol v2: mã hoá nhị phân viết tay ở hai phía
# là hai bản dịch của cùng một tài liệu, và chúng lệch nhau TRONG IM LẶNG. Client đọc lệch
# một byte thì không có lỗi nào được ném ra — nó chỉ đọc rác, rồi nhân vật nhảy lung tung
# hoặc màn hình đứng. Đã dính đúng một lần với trường `mood` (xem netproto.ProtocolVersion).
#
# `cd server && go run ./cmd/protogen` đọc file này và sinh ra:
#   server/internal/wire/wire_gen.go   (Go: mã hoá + giải mã)
#   client/src/net/wire.gen.ts         (TypeScript: giải mã + mã hoá)
# ĐỪNG sửa tay hai file đó — sửa ở đây rồi chạy lại generator.
#
# ── Cách viết ───────────────────────────────────────────────────────────────────────────
#   message <Tên> <id>      bắt đầu một message; `id` đi vào byte đầu của gói
#   doc <câu>               mô tả, viết bao nhiêu dòng `doc` cũng được
#   <tên> <kiểu>            một trường
#   <tên> <kiểu> bitmask a,b,c    trường này là bitmask điều khiển các trường a, b, c
#   <tên> <kiểu> if a             trường này CHỈ có mặt khi bit `a` bật
#
# Kiểu: u8 u16 u32 i8 i16 i32 f32 str
#   str = 1 byte độ dài + UTF-8 (tối đa 255 byte)
#   mọi số nguyên là little-endian
#
# ⚠ `id` của message và THỨ TỰ các trường là một phần của giao thức. Đổi chúng khi đã có
#   người chơi đang kết nối nghĩa là mọi client cũ đọc sai. Thêm trường mới vào CUỐI.

version 2

message Spawn 1
  doc Entity vào vùng quan tâm lần đầu, hoặc quay lại sau khi đã Despawn (§4.3).
  doc Chở ĐỦ descriptor + state nên client dựng được ngay, không phải hỏi thêm.
  id          u32
  desc_ver    u16     # phiên bản descriptor; đổi thì server gửi Desc
  name        str
  char        u8      # chỉ số trong config.characters
  title       u8
  x           u16     # lượng tử theo khu: 12000/65535 ≈ 0.18 đơn vị (§4.6)
  y           u16
  facing      u8      # góc 0..255 ứng với 0..2π
  level       u8
  scale       u8      # bước 0.02
  swords      u16
  ring_phase  u16
  ring_dir    i8
  mood        u8
  buffs       u16     # bitmask buff
  flags       u16     # bitmask cờ trạng thái
  tilt        i8      # độ nghiêng lưỡi kiếm, bước 1/128 rad (|tilt| ≤ 35° = 0.61 rad)
  status      u16     # bitmask trạng thái do kỹ năng (ngủ, trói, choáng…) — MỌI người thấy
  crack       u8      # vết nứt vòng kiếm 0..1, bước 1/255

message Update 3
  doc Delta mỗi snapshot: chỉ chở trường ĐÃ ĐỔI so với bản client đã ack (§4.3).
  doc Đây là chỗ tiết kiệm băng thông lớn nhất — đứng yên thì gói gần như rỗng.
  id          u32
  changed     u16     bitmask x,y,facing,level,scale,swords,ring_phase,ring_dir,mood,buffs,flags,tilt,status,crack
  x           u16     if x
  y           u16     if y
  facing      u8      if facing
  level       u8      if level
  scale       u8      if scale
  swords      u16     if swords
  ring_phase  u16     if ring_phase
  ring_dir    i8      if ring_dir
  mood        u8      if mood
  buffs       u16     if buffs
  flags       u16     if flags
  tilt        i8      if tilt
  status      u16     if status
  crack       u8      if crack

message Despawn 4
  doc Entity rời đi THẬT SỰ — thứ bước 2b còn thiếu.
  doc Không có message này thì client phải đoán bằng cách đợi 2 giây im lặng, nên không bao
  doc giờ phân biệt được "vắng mặt vì AOI" với "đã đi mất".
  doc reason: 0 out_of_view · 1 died · 2 teleported · 3 left
  id          u32
  reason      u8

message SelfState 5
  doc Trạng thái của CHÍNH MÌNH, giữ độ chính xác cao (§4.6).
  doc Vị trí người khác lượng tử về u16 thì không ai thấy; vị trí của chính mình dùng để
  doc hiệu chỉnh dự đoán (GAME_SPEC §8.2a) nên sai số lượng tử ở đây biến thành giật.
  ack_seq     u32     # input cuối server đã xử lý
  x           i32     # fixed-point 1/256 đơn vị
  y           i32
  vx          i32
  vy          i32

message Input 6
  doc Client → server, 30 lần/giây. Không có trường nào thừa.
  seq         u32
  angle       u8
  buttons     u8      # bit0 moving · bit1 burn · bit2 skill

message Event 7
  doc Sự kiện trong khu (chém trúng, nhặt đồ, đột phá, kỹ năng…) — §4.1.
  doc
  doc Đây là phần NẶNG NHẤT của gói: 37% ở bản msgpack v1, vì mỗi sự kiện chở `kind` dạng
  doc CHUỖI ("TienKhiPicked" = 14 byte) cộng tên khoá của từng trường. Ở đây `kind` là 1
  doc byte tra bảng `EventKinds`, và `has` quyết trường nào có mặt — sự kiện thường chỉ
  doc dùng 2–3 trường nên phần còn lại không tốn byte nào.
  kind        u8      # chỉ số trong netproto.EventKinds
  has         u8      bitmask a,b,n,s,f
  x           u16     # lượng tử theo khu như vị trí người chơi
  y           u16
  a           u32     if a
  b           u32     if b
  n           i32     if n
  s           str     if s
  f           f32     if f

message ItemAdd 8

  id          u32
  x           u16
  y           u16
  kind        u8      # chỉ số trong netproto.ItemKinds
  has         u8      bitmask value
  value       f32     if value

message Monster 10
  doc Quái trong tầm nhìn. Phần tĩnh (loại, máu tối đa, bán kính) chỉ gửi LẦN ĐẦU, và máu
  doc chỉ gửi khi ĐỔI — ba bit riêng thay vì một bit "static" chung, vì generator đòi mỗi
  doc bit của bitmask phải là một trường có thật (và cách đó cũng gọn hơn: quái đứng yên
  doc không đổi máu thì gói chỉ còn id + x + y + has = 9 byte).
  id          u32
  x           u16
  y           u16
  has         u8      bitmask hp,kind,max_hp,r
  hp          u16     if hp
  kind        u8      if kind
  max_hp      u16     if max_hp
  r           u16     if r

message MonsterDel 11
  doc Quái rời tầm nhìn hoặc chết.
  id          u32

message SnapHead 12
  doc Đầu mỗi snapshot. Sau nó là một dãy message, mỗi message mở đầu bằng 1 byte loại.
  doc
  doc `ack` là seq input cuối server đã xử lý — client dùng để biết delta tính từ đâu
  doc (§4.3). `flags` bit0 = keyframe: gói này chở TOÀN BỘ trạng thái, không phải delta.
  tick        u32
  ack         u32
  time_ms     u32     # đồng hồ server, ms kể từ lúc phòng mở
  srv_us      u16     # thời gian xử lý một tick, micro-giây (chỉ để đo §13.6)
  flags       u8      # bit0 keyframe

message SelfExtra 13
  doc Phần trạng thái CHỈ chủ nhân cần: EXP, lượt kỹ năng, danh hiệu, Tiên khí.
  doc
  doc Tách khỏi SelfState vì SelfState đi MỌI tick (vị trí để hiệu chỉnh dự đoán) còn mấy
  doc số này gần như không đổi. `status` và `crack` KHÔNG ở đây: chúng ở Spawn/Update vì
  doc người khác cũng phải thấy — client vẽ "zzz" trên đầu người đang ngủ.
  has         u8      bitmask exp,charges,title,tien_have,tien_need
  exp         f32     if exp
  charges     u8      if charges
  title       u8      if title
  tien_have   f32     if tien_have
  tien_need   f32     if tien_need

message LbHead 14
  doc Đầu gói BẢNG XẾP HẠNG PHÒNG, 1 lần/giây (GAME_SPEC §11.5, v6.19). Sau nó là `count`
  doc message LbRow, mỗi cái mở đầu bằng 1 byte loại — y như SnapHead.
  doc
  doc `rank` là hạng THẬT của người nhận trong cả phòng (5 khu) — để ghim dòng "Bạn #14"
  doc khi mình ngoài top 10. 0 = chưa có hạng (vừa vào, đang hồi sinh).
  rank        u16
  total       u16     # số người (thật + bot) cả phòng đang được xếp
  my_score    u32     # Uy Danh của người nhận, làm tròn
  count       u8      # số LbRow theo sau (≤ 10)

message LbRow 15

  doc gửi khu thật rồi để client tự viết "???" thì ai mở DevTools cũng đọc được.
  id          u32
  name        str
  character   u8      # chỉ số trong config characters; 255 = không rõ
  title_id    u8      # danh hiệu đang đeo (1–10, đã qua chống nhấp nháy 3 s); 0 = chưa có
  rank        u8      # hạng 1–10
  zone_id     u8      # 0–4; 255 = Ẩn Thân
  realm       u8      # 0–8 phàm, 9+ Tiên giới
  tier        u8      # tầng 1–10
  score       u32     # Uy Danh, làm tròn
  stealth     u8      # 1 = đang Ẩn Thân
